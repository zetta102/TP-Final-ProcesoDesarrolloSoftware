== Paso 1: Finalizar Scrim ==
Org -> API : POST /v1/api/scrims/{id}/finalizar
API -> Facade : finishScrim(id)
Facade -> Service : finishScrimById(scrimId)

Service -> DB : scrimRepository.getReferenceById(scrimId)
DB --> Service : scrim
Service -> Service : lobby = scrim.getLobbyId()
Service -> Context : new ScrimContext(lobby, PlayingState, eventPublisher)
Service -> Context : finalizar()
Context -> State : finalizar(ctx)
State -> Context : setState(new FinishedState())
note right: Lobby status → "Finalizado"\nSe publica ScrimStateChangedEvent

Service -> DB : lobbyRepository.save(lobby)
Service -> ScrimModel : scrim.setStatus("Finalizado")
Service -> DB : scrimRepository.save(scrim)
Service -> Stats : stats.setEndTime(now)
Service -> Stats : stats.setStatus("Finalizado")
Service -> DB : scrimStatisticsRepository.save(stats)

Service --> Facade : "Scrim finalizado exitosamente."
Facade --> API : mensaje
API --> Org : 200 OK

API -> Facade : saveStatistics(id, request)

== Paso 3: Recalcular MMR ==
Service -> Recalculator : recalculate(playerStatsList, winningTeam, redTeam, blueTeam)

loop Para cada PlayerScrimStats

    Recalculator -> Recalculator : recalcular KDA (promedio móvil)
